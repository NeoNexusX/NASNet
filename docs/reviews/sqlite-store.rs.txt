//! SQLite 配置存储。完整替换在事务内完成，加载后重新执行模型校验。

use std::{fmt, path::Path};

use rusqlite::Connection;

use crate::{config::{ConfigIoError, encode_device,decode_device}, model::Device};

/// 存储整个设备列表；列表顺序属于配置的一部分。
pub trait DeviceStore {
    fn save_all(&mut self, devices: &[Device]) -> Result<(), StoreError>;
    fn load_all(&self) -> Result<Vec<Device>, StoreError>;
    fn delete_device(&mut self, id: &str) -> Result<bool, StoreError>;
}

pub struct SqliteStore {
    connection: Connection,
}

impl SqliteStore {
    pub fn open(path: &Path) -> Result<Self, StoreError> {
        let con = Connection::open(path)?;
        con.execute_batch(
            "CREATE TABLE IF NOT EXISTS devices (
                id TEXT PRIMARY KEY NOT NULL,
                position INTEGER NOT NULL UNIQUE CHECK (position >= 0),
                config_json TEXT NOT NULL);"
        )?;
        Ok(Self { connection: con })
    }
}

impl DeviceStore for SqliteStore {
    fn save_all(&mut self, devices: &[Device]) -> Result<(), StoreError> {
        let tx = self.connection.transaction()?;
        tx.execute("DELETE FROM devices", [])?; 
        for (i, device) in devices.iter().enumerate() {
            let position = i64::try_from(i)?;
            let config_json = encode_device(device)?;
            tx.execute(
                "INSERT INTO devices (id, position, config_json) VALUES (?1, ?2, ?3)",
                rusqlite::params![device.id(), position, config_json],
            )?;
        }
        tx.commit()?;
        Ok(())
    }

    fn load_all(&self) -> Result<Vec<Device>, StoreError> {
        let mut query_a = self.connection.prepare( "SELECT id, config_json FROM devices ORDER BY position ASC")?;
        let query_b = query_a.query_map([], |row| {
            let id: String = row.get(0)?;
            let config_json: String = row.get(1)?;
            Ok((id, config_json))
        })?;
        let mut temp = Vec::<Device>::new();
        for result in query_b {
            let (id, config_json) = result?;
            let device = decode_device(&config_json)?;
            if device.id() != id {
                return Err(StoreError::DeviceIdMismatch);
            }
            temp.push(device);
        }
        Ok(temp)
    }

    fn delete_device(&mut self, id: &str) -> Result<bool, StoreError> {
        let result = self.connection.execute("DELETE FROM devices WHERE id = ?1", [id])? > 0;
        Ok(result)
    }
}

#[derive(Debug)]
pub enum StoreError {
    Database(rusqlite::Error),
    Config(ConfigIoError),
    Position(std::num::TryFromIntError),
    // TODO 4.3 完成后移除临时豁免。
    #[allow(dead_code)]
    DeviceIdMismatch,
}

impl fmt::Display for StoreError {
    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Database(error) => write!(formatter, "数据库操作失败：{error}"),
            Self::Config(error) => write!(formatter, "存储配置无效：{error}"),
            Self::Position(error) => write!(formatter, "设备位置超出数据库范围：{error}"),
            Self::DeviceIdMismatch => write!(formatter, "数据库设备 ID 与配置内 ID 不一致"),
        }
    }
}

impl std::error::Error for StoreError {
    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
        match self {
            Self::Database(error) => Some(error),
            Self::Config(error) => Some(error),
            Self::Position(error) => Some(error),
            Self::DeviceIdMismatch => None,
        }
    }
}

impl From<rusqlite::Error> for StoreError {
    fn from(error: rusqlite::Error) -> Self {
        Self::Database(error)
    }
}

impl From<ConfigIoError> for StoreError {
    fn from(error: ConfigIoError) -> Self {
        Self::Config(error)
    }
}

impl From<std::num::TryFromIntError> for StoreError {
    fn from(error: std::num::TryFromIntError) -> Self {
        Self::Position(error)
    }
}
